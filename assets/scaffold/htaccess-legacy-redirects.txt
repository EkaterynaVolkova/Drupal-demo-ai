# Legacy URLs of the old shapovalov.md CMS (301 to the current /rus/<cpu> aliases).
<IfModule mod_rewrite.c>
  RewriteEngine on
  # /rus/<cpu>/ -> /rus/<cpu>
  RewriteRule ^rus/(.+[^/])/+$ /rus/$1 [R=301,L]
  # /rus, /rom, /eng and their language roots -> front page
  RewriteRule ^(?:rus|rom|eng)/?$ / [R=301,L]
  # /rom/<cpu> and /eng/<cpu> -> Russian page (the only migrated language)
  RewriteRule ^(?:rom|eng)/(.+[^/])/*$ /rus/$1 [R=301,L]
  # /index.php?lang=rus&content=<cpu> -> /rus/<cpu>
  RewriteCond %{QUERY_STRING} (?:^|&)content=([^&]+)
  RewriteRule ^index\.php$ /rus/%1? [R=301,L]
</IfModule>
